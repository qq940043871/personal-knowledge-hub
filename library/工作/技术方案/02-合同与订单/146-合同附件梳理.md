```java
DeliveryRule matchedRule = null;

for (DeliveryRule rule : deliveryRules) {
    boolean isMatch = true;

    // 必选匹配：交收类型
    if (!rule.getDeliveryType().equals(context.getDeliveryType())) {
        isMatch = false;
    }

    // 必选匹配：担保类型
    if (!rule.getGuaranteeType().contains(context.getGuaranteeType())) {
        isMatch = false;
    }

    // 可选匹配：如果规则里配置了申请方式，才进行匹配
    if (rule.getApplyType() != null 
        && !rule.getApplyType().equals(context.getApplyType())) {
        isMatch = false;
    }

    // 可选匹配：配货方式（同理）
    if (rule.getAllocateType() != null 
        && !rule.getAllocateType().equals(context.getAllocateType())) {
        isMatch = false;
    }

    // 最终匹配成功
    if (isMatch) {
        matchedRule = rule;
        break;
    }
}

if (matchedRule == null) {
    throw new IllegalArgumentException("未找到匹配的交收规则");
}
```
