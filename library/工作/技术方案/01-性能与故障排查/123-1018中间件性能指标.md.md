# 1018中间件性能指标

//日志格式 中间件|开始时间|协议号|耗时|结果|原因|
-- 10.168.15.106_01|2020-09-16 00:00:00|K00001|16|000000|

//统计当日总协议数
cat mointor.20200915.log|wc -l

//分协议号统计调用次数、并排序
cat monintor.20200915.log|gawk -F'|' '{print 34}'|gawk '{sum[$1]+=}END{for(i in sum)print i "\t" sum[i]}'|sort -nr -k 2

//统计每秒钟协议调用次数
cat monintor.20200915.log|gawk -F'|' '{print 23 4}'|gawk '{sum[$2,$3]+=}END{for(i in sum)print i "\t" sum[i]}'|awk '4}'|gawk '{sum[$2,$3]+=}END{for(i in sum)print i "\t" sum[i]}'|awk '2=="PA0002"{print $0}'|sort -n -k 1

//统计排名前20的协议

cat time.txt |gawk -F 'Mid协议号：' '{print $2}'|gawk -F '花费时间:' '{print $1}'|sort|uniq -c|sort -rn|head -n 20
