### 1. Edge-TTS

**特点**：微软技术、免费、中文好、无需GPU

```
<code>pip install edge-tts</code>
```

```
<code>import asyncio<br/>import edge_tts<br/><br/>async def text_to_speech():<br/>    """文本转语音"""<br/>    # 合成<br/>    communicate = edge_tts.Communicate(<br/>        text="你好，这是一段测试语音。",<br/>        voice="zh-CN-XiaoxiaoNeural"  # 中文女声<br/>    )<br/>    await communicate.save("output.mp3")<br/><br/>    # 男声<br/>    communicate = edge_tts.Communicate(<br/>        text="你好，这是男声。",<br/>        voice="zh-CN-YunxiNeural"<br/>    )<br/>    await communicate.save("male.mp3")<br/><br/>    # 不同风格<br/>    # 新闻: zh-CN-XiaoxiaoNeural<br/>    # 客服: zh-CN-XiaoyiNeural<br/>    # 活泼: zh-CN-liaoning-XiaobaiNeural<br/>    # 成熟: zh-CN-shaanxi-XiaosiNeural<br/><br/>async def long_text():<br/>    """长文本合成(带进度)"""<br/>    text = """<br/>    这是一段较长的文本。语音合成技术正在改变人机交互的方式。<br/>    通过深度学习模型，我们可以生成自然流畅的语音。<br/>    这项技术广泛应用于智能助手、导航系统、有声读物等领域。<br/>    """<br/><br/>    communicate = edge_tts.Communicate(text, voice="zh-CN-XiaoxiaoNeural")<br/>    await communicate.save("long_text.mp3")<br/><br/>async def get_voices():<br/>    """获取所有可用音色"""<br/>    voices = await edge_tts.list_voices()<br/>    for voice in voices:<br/>        if voice["Locale"].startswith("zh"):<br/>            print(f"{voice['Name']} - {voice['Gender']} - {voice['Locale']}")<br/><br/># 运行<br/>asyncio.run(text_to_speech())</code>
```

### 2. gTTS (Google TTS)

**特点**：简单易用、Google翻译接口、免费

```
<code>pip install gtts</code>
```

```
<code>from gtts import gTTS<br/>import playsound<br/><br/># 基础合成<br/>tts = gTTS(text="你好世界", lang="zh-CN")<br/>tts.save("hello.mp3")<br/><br/># 播放<br/>playsound.playsound("hello.mp3")<br/><br/># 英文合成<br/>tts = gTTS(text="Hello, this is a test.", lang="en")<br/>tts.save("hello_en.mp3")<br/><br/># 慢速<br/>tts = gTTS(text="Hello", lang="en", slow=True)<br/>tts.save("hello_slow.mp3")<br/><br/># 批量文本<br/>texts = ["第一段", "第二段", "第三段"]<br/>for i, text in enumerate(texts):<br/>    tts = gTTS(text=text, lang="zh-CN")<br/>    tts.save(f"part_{i}.mp3")</code>
```

### 3. pyttsx3 (离线)

**特点**：完全离线、不需要网络、系统自带

```
<code>pip install pyttsx3</code>
```

```
<code>import pyttsx3<br/><br/># 初始化<br/>engine = pyttsx3.init()<br/><br/># 获取语音列表<br/>voices = engine.getProperty('voices')<br/>for voice in voices:<br/>    print(f"ID: {voice.id}, Name: {voice.name}, Languages: {voice.languages}")<br/><br/># 设置语音<br/>engine.setProperty('voice', voices[0].id)  # 选择第一个语音<br/><br/># 设置参数<br/>engine.setProperty('rate', 150)    # 语速(默认200)<br/>engine.setProperty('volume', 0.9)  # 音量(0-1)<br/><br/># 合成<br/>engine.say("你好，这是一段离线语音合成测试。")<br/>engine.say("This is an offline TTS test.")<br/>engine.runAndWait()<br/><br/># 保存为文件<br/>engine.save_to_file("output.wav", "hello")<br/>engine.runAndWait()<br/><br/># Windows使用SAPI5，Linux使用espeak</code>
```

```
<code># （伪代码骨架，示意 VoiceSynthesizer 封装思路；实现细节按需展开）<br/>import asyncio<br/>import edge_tts<br/>from pydub import AudioSegment<br/><br/>class VoiceSynthesizer:<br/>    """Edge-TTS 合成 + pydub 后处理的封装（示意）"""<br/><br/>    def __init__(self):<br/>        self.voices = {<br/>            'xiaoxiao': 'zh-CN-XiaoxiaoNeural',  # 女声<br/>            'yunxi': 'zh-CN-YunxiNeural',        # 男声<br/>            'en_female': 'en-US-JennyNeural',<br/>        }<br/><br/>    async def synthesize(self, text, output_path, voice='xiaoxiao',<br/>                         pitch='+0%', rate='+0%'):<br/>        # ★特殊点1 edge_tts 是异步 API：Communicate(...).save() 是协程，必须 await<br/>        communicate = edge_tts.Communicate(<br/>            text=text, voice=self.voices.get(voice, self.voices['xiaoxiao']),<br/>            pitch=pitch, rate=rate)<br/>        await communicate.save(output_path)<br/><br/>    def synthesize_sync(self, text, output_path, voice='xiaoxiao'):<br/>        # 同步入口：包一层 asyncio.run，方便脚本直接调用<br/>        asyncio.run(self.synthesize(text, output_path, voice))<br/><br/>    def merge_audio(self, audio_files, output_path, fade_duration=500):<br/>        # ★特殊点2 多段拼接配 fade_in/fade_out 淡入淡出，避免接缝爆音<br/>        combined = AudioSegment.empty()<br/>        for i, file in enumerate(audio_files):<br/>            audio = AudioSegment.from_mp3(file)<br/>            if i &gt; 0:<br/>                audio = audio.fade_in(fade_duration)<br/>            if i &lt; len(audio_files) - 1:<br/>                audio = audio.fade_out(fade_duration)<br/>            combined += audio<br/>        combined.export(output_path, format='mp3')<br/><br/>    def adjust_speed(self, audio_path, output_path, speed_factor=1.0):<br/>        # ★特殊点3 变速靠 _spawn 重设 frame_rate 再回 44100（依赖 pydub 内部行为的常用技巧）<br/>        audio = AudioSegment.from_mp3(audio_path)<br/>        faster = audio._spawn(audio.raw_data, overrides={<br/>            'frame_rate': int(audio.frame_rate * speed_factor)})<br/>        faster.set_frame_rate(44100).export(output_path, format='mp3')<br/><br/># 使用：synthesizer = VoiceSynthesizer()<br/># asyncio.run(synthesizer.synthesize("欢迎收听今天的新闻。", "part_0.mp3"))<br/># synthesizer.merge_audio(["part_0.mp3", "part_1.mp3"], "news.mp3")<br/># synthesizer.adjust_speed("news.mp3", "news_fast.mp3", 1.2)</code>
```
