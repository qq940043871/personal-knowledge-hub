```
<code>pip install opencv-python</code>
```

```
<code>import cv2<br/>import numpy as np<br/>from PIL import Image<br/><br/># ============ 视频读取 ============<br/>cap = cv2.VideoCapture('input.mp4')<br/><br/># 获取视频属性<br/>fps = cap.get(cv2.CAP_PROP_FPS)<br/>width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))<br/>height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))<br/>frame_count = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))<br/><br/>print(f"FPS: {fps}, 分辨率: {width}x{height}, 总帧数: {frame_count}")<br/><br/># 逐帧读取<br/>frames = []<br/>while cap.isOpened():<br/>    ret, frame = cap.read()<br/>    if not ret:<br/>        break<br/>    frames.append(frame)<br/>cap.release()<br/><br/>print(f"读取了 {len(frames)} 帧")<br/><br/># ============ 视频写入 ============<br/>fourcc = cv2.VideoWriter_fourcc(*'mp4v')  # 或 'XVID', 'MJPG'<br/>out = cv2.VideoWriter('output.mp4', fourcc, 30.0, (640, 480))<br/><br/>for i in range(100):<br/>    frame = np.zeros((480, 640, 3), dtype=np.uint8)<br/>    frame[:] = (0, 255, 0)  # 绿色背景<br/>    cv2.putText(frame, f"Frame {i}", (50, 240), <br/>                 cv2.FONT_HERSHEY_SIMPLEX, 2, (255, 255, 255), 3)<br/>    out.write(frame)<br/><br/>out.release()<br/><br/># ============ 视频处理 ============<br/># 视频缩放<br/>cap = cv2.VideoCapture('input.mp4')<br/>out = cv2.VideoWriter('resized.mp4', fourcc, 30.0, (320, 240))<br/><br/>while True:<br/>    ret, frame = cap.read()<br/>    if not ret:<br/>        break<br/>    resized = cv2.resize(frame, (320, 240))<br/>    out.write(resized)<br/><br/>cap.release()<br/>out.release()<br/><br/># ============ 抽帧 ============<br/>cap = cv2.VideoCapture('input.mp4')<br/>frame_idx = 0<br/>saved_idx = 0<br/><br/>while True:<br/>    ret, frame = cap.read()<br/>    if not ret:<br/>        break<br/><br/>    # 每30帧保存一张图<br/>    if frame_idx % 30 == 0:<br/>        cv2.imwrite(f'frame_{saved_idx:04d}.jpg', frame)<br/>        saved_idx += 1<br/><br/>    frame_idx += 1<br/><br/>cap.release()</code>
```

### 3. Stable Video Diffusion (SVD)

**特点**：图片转视频、扩散模型驱动

```
<code>pip install torch torchvision<br/>pip install diffusers transformers</code>
```

```
<code>from diffusers import StableVideoDiffusionPipeline<br/>from diffusers.utils import load_image<br/>import torch<br/>from PIL import Image<br/>import cv2<br/><br/># 加载模型<br/>pipe = StableVideoDiffusionPipeline.from_pretrained(<br/>    "stabilityai/stable-video-diffusion-img2vid-xt",<br/>    torch_dtype=torch.float16,<br/>    variant="fp16"<br/>)<br/>pipe = pipe.to("cuda")<br/><br/># 加载并预处理图像<br/>image = load_image("input_image.jpg")<br/>image = image.resize((1024, 576))  # 16:9比例<br/><br/># 生成视频(25帧)<br/>generator = torch.Generator(device="cuda").manual_seed(42)<br/>frames = pipe(<br/>    image,<br/>    num_frames=25,<br/>    decode_chunk_size=8,<br/>    generator=generator,<br/>    num_inference_steps=25,<br/>    guidance_scale=3.0<br/>).frames[0]<br/><br/># 保存视频<br/>output_path = "output_video.mp4"<br/>height, width = frames[0].shape[:2]<br/><br/>fourcc = cv2.VideoWriter_fourcc(*'mp4v')<br/>video = cv2.VideoWriter(output_path, fourcc, fps=8, frameSize=(width, height))<br/><br/>for frame in frames:<br/>    video.write(cv2.cvtColor(np.array(frame), cv2.COLOR_RGB2BGR))<br/><br/>video.release()</code>
```
