## 推理接口：同步 / 异步 / 流式三种生成方式

> `InferenceEngine` 持有 model、tokenizer、config 与 **kv_cache**，对外提供 **generate（同步）、generate_async（异步）、generate_stream（流式）** 三种生成方式。

```python
class InferenceEngine:
    """推理引擎核心"""
    
    def __init__(self, model: LLMModel, tokenizer: Tokenizer, config: InferenceConfig):
        self.model = model
        self.tokenizer = tokenizer
        self.config = config
        self.kv_cache = {}
    
    def generate(self, prompt: str, **kwargs) -> str:
        """同步生成"""
        inputs = self.tokenizer.encode(prompt, return_tensors="pt")
        outputs = self.model.generate(**inputs, **kwargs)
        return self.tokenizer.decode(outputs[0], skip_special_tokens=True)
    
    async def generate_async(self, prompt: str, **kwargs) -> str:
        """异步生成"""
        inputs = self.tokenizer.encode(prompt, return_tensors="pt")
        outputs = await asyncio.to_thread(
            self.model.generate, **inputs, **kwargs
        )
        return self.tokenizer.decode(outputs[0], skip_special_tokens=True)
    
    def generate_stream(self, prompt: str, **kwargs):
        """流式生成"""
        inputs = self.tokenizer.encode(prompt, return_tensors="pt")
        for output in self.model.generate_stream(**inputs, **kwargs):
            token = self.tokenizer.decode([output], skip_special_tokens=True)
            yield token
```

## 生成参数：GenerateConfig 的采样与停止条件

参数分三组：**生成长度与采样核心**（max_tokens、temperature、top_p、top_k、repetition_penalty、length_penalty、num_beams）、**停止条件**（stop_strings、stop_tokens）、**采样开关**（do_sample、epsilon_cutoff、eta_cutoff）。

```python
@dataclass
class GenerateConfig:
    max_tokens: int = 256
    temperature: float = 0.7
    top_p: float = 0.9
    top_k: int = 50
    repetition_penalty: float = 1.0
    length_penalty: float = 1.0
    num_beams: int = 1
    
    # 停止条件
    stop_strings: List[str] = None
    stop_tokens: List[int] = None
    
    # 采样
    do_sample: bool = True
    epsilon_cutoff: float = None
    eta_cutoff: float = None
```

## 采样策略：Greedy / Random / Top-K / Top-P 四种采样

```python
class Sampler:
    """采样策略"""
    
    @staticmethod
    def greedy(logits: Tensor) -> int:
        """贪婪采样"""
        return logits.argmax(-1).item()
    
    @staticmethod
    def random(logits: Tensor, temperature: float = 1.0) -> int:
        """随机采样"""
        probs = F.softmax(logits / temperature, dim=-1)
        return torch.multinomial(probs, 1).item()
    
    @staticmethod
    def top_k(logits: Tensor, k: int, temperature: float = 1.0) -> int:
        """Top-K采样"""
        values, indices = torch.topk(logits, k)
        probs = F.softmax(values / temperature, dim=-1)
        return indices[torch.multinomial(probs, 1)].item()
    
    @staticmethod
    def top_p(logits: Tensor, p: float, temperature: float = 1.0) -> int:
        """Top-P（核）采样"""
        probs = F.softmax(logits / temperature, dim=-1)
        sorted_probs, sorted_indices = torch.sort(probs, descending=True)
        cumsum = torch.cumsum(sorted_probs, dim=-1)
        
        # 找到满足概率阈值的最小集合
        mask = cumsum <= p
        mask[..., -1] = True  # 至少选一个
        
        valid_indices = sorted_indices[mask]
        valid_probs = sorted_probs[mask]
        
        return valid_indices[torch.multinomial(valid_probs, 1)].item()
```
