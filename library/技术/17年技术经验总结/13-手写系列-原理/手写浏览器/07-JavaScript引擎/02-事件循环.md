## 一、事件循环模型

```
事件循环：
  while (true) {
    // 1. 从宏任务队列取出 1 个任务执行
    //    （页面首次运行时，同步脚本本身作为第一个宏任务）
    task = macrotaskQueue.dequeue();
    execute(task);

    // 2. 清空微任务队列（执行中新增的微任务也在本轮处理完）
    while (!microtaskQueue.empty()) {
      execute(microtaskQueue.dequeue());
    }

    // 3. 渲染阶段：每帧至多一次，是否渲染由浏览器决定
    //    rAF 回调与 UI 渲染属于渲染帧步骤，不是宏任务
    if (needsRendering) {
      runRequestAnimationFrameCallbacks();
      updateRendering();
    }
  }

微任务：
  - Promise.then/catch/finally
  - queueMicrotask()
  - MutationObserver

宏任务：
  - setTimeout
  - setInterval
  - I/O操作
  - UI事件回调（click、input 等事件派发）
```

## 二、C++实现

```cpp
class EventLoop {
private:
    std::queue<std::function<void()>> taskQueue;
    std::queue<std::function<void()>> microtaskQueue;
    std::queue<std::function<void()>> animationCallbacks;
    bool running = false;
    
public:
    void run() {
        running = true;
        
        while (running) {
            // 1. 处理任务队列
            processTasks();
            
            // 2. 处理微任务
            processMicrotasks();
            
            // 3. 处理动画回调
            processAnimationCallbacks();
            
            // 4. 等待下一个任务
            if (taskQueue.empty() && microtaskQueue.empty()) {
                waitForEvents();
            }
        }
    }
    
    void processTasks() {
        while (!taskQueue.empty()) {
            std::function<void()> task = taskQueue.front();
            taskQueue.pop();
            task();
        }
    }
    
    void processMicrotasks() {
        while (!microtaskQueue.empty()) {
            std::function<void()> task = microtaskQueue.front();
            microtaskQueue.pop();
            task();
        }
    }
    
    void enqueueTask(std::function<void()> task) {
        taskQueue.push(task);
    }
    
    void enqueueMicrotask(std::function<void()> task) {
        microtaskQueue.push(task);
    }
};

class Timer {
public:
    static void setTimeout(std::function<void()> callback, int delay) {
        uint32_t id = nextId++;
        timers[id] = {callback, delay, getCurrentTime() + delay};
        eventLoop.enqueueTask([id](.) {
            if (timers[id].callback) {
                timers[id].callback();
            }
            timers.erase(id);
        });
    }
    
    static void setInterval(std::function<void()> callback, int interval) {
        uint32_t id = nextId++;
        timers[id] = {callback, interval, getCurrentTime() + interval, true};
    }
};
```
